def test_live_casimir_potential_and_hostile_bites():
    def log_eta(tau):
        q = np.exp(2j * np.pi * tau)
        prod = 1.0 + 0j
        qn = q
        for n in range(1, 100):
            prod *= (1.0 - qn)
            qn *= q
            if abs(qn) < 1e-15: break
        return 1j * np.pi * tau / 12.0 + np.log(prod)

    def V_casimir(tau):
        y = tau.imag
        return float(- (np.log(y) + 4.0 * log_eta(tau).real))

    omega = complex(-0.5, np.sqrt(3)/2.0)
    tau_i = 1j

    v_om = V_casimir(omega)
    v_i = V_casimir(tau_i)
    v_cusp = V_casimir(4j)

    # Minimum at omega
    assert abs(v_om - 1.033519) < 1e-4
    # Saddle point at i has higher energy
    assert v_i > v_om + 0.02
    # Cusp diverges to +infinity
    assert v_cusp > v_om + 1.5


def test_live_green_function_and_12_point_orbit():
    omega = complex(-0.5, np.sqrt(3)/2.0)
    
    def th1(u, t, NK=25):
        k = np.arange(-NK, NK + 1)
        h = k + 0.5
        return -1j * np.sum((-1.0)**k * np.exp(1j*np.pi*t*h*h + 1j*np.pi*(2*k + 1)*u))

    def log_eta(tau):
        q = np.exp(2j * np.pi * tau)
        prod = 1.0 + 0j
        qn = q
        for n in range(1, 80):
            prod *= (1.0 - qn)
            qn *= q
            if abs(qn) < 1e-15: break
        return 1j * np.pi * tau / 12.0 + np.log(prod)

    eta_omega = np.exp(log_eta(omega))

    def green_func(w, t=omega):
        val = th1(w, t)
        if abs(val) < 1e-12: return 1e10
        return float(- np.log(abs(val / eta_omega)**2) + 2 * np.pi * (w.imag**2) / t.imag)

    all_72 = [(a, b) for a in range(9) for b in range(9) if (a % 3 != 0 or b % 3 != 0)]
    g_vals = {}
    for a, b in all_72:
        w = (a/9.0) * omega - (b/9.0)
        g_vals[(a, b)] = round(green_func(w), 5)

    min_g = min(g_vals.values())
    ground_pts = [pt for pt, val in g_vals.items() if abs(val - min_g) < 1e-4]
    
    # Exactly 12 points in ground state
    assert len(ground_pts) == 12
    # Minimum energy is negative
    assert min_g < -0.50


def test_live_cp_odd_dominance():
    # 12 ground state points
    pts = [(2, 3), (2, 4), (3, 2), (3, 4), (4, 2), (4, 3),
           (5, 6), (5, 7), (6, 5), (6, 7), (7, 5), (7, 6)]

    def mul_w(x, y):
        a, b = x; c, d = y
        return ((a*c - b*d) % 9, (a*d + b*c - b*d) % 9)

    def conj_w(x):
        a, b = x
        return ((a - b) % 9, (-b) % 9)

    units_6 = [(1, 0), (0, 1), (-1, -1), (-1, 0), (0, -1), (1, 1)]

    def is_cp_even(pt):
        pt_bar = conj_w(pt)
        for u in units_6:
            if mul_w(u, pt_bar) == pt:
                return True
        return False

    cp_even = sum(is_cp_even(pt) for pt in pts)
    cp_odd = len(pts) - cp_even

    # Hostile check: 2:1 dominance of CP-odd points
    assert cp_odd == 8
    assert cp_even == 4
